import face_recognition
import cv2
import os
import glob
import numpy as np

# 1. Load sample faces and their names automatically
known_face_encodings = []
known_face_names = []

# Looks inside the employees folder for any image format
images_path = 'employees/*'
found_images = glob.glob(images_path)

print(f"Loading {len(found_images)} employee profiles...")

for img_path in found_images:
    try:
        # ---> THE FIX: Use OpenCV to forcefully read the image instead of face_recognition <---
        img_bgr = cv2.imread(img_path)
        
        # Check if OpenCV failed to read the file entirely
        if img_bgr is None:
            print(f"❌ CRITICAL ERROR: Cannot read {img_path}. The file is corrupted, empty, or not a real image.")
            continue
            
        # Convert OpenCV's default BGR color space to RGB
        img_rgb = cv2.cvtColor(img_bgr, cv2.COLOR_BGR2RGB)
        
        # Pass the newly converted RGB image to get encodings
        encodings = face_recognition.face_encodings(img_rgb)
        
        if len(encodings) > 0:
            known_face_encodings.append(encodings[0])
            # Extracts the name from the file (e.g., 'kurt' from 'employees/kurt.jpg')
            name = os.path.splitext(os.path.basename(img_path))[0].replace('_', ' ').title()
            known_face_names.append(name)
            print(f"-> Successfully loaded profile for: {name}")
        else:
            print(f"⚠️ Warning: No face detected in {img_path}. Skipping.")
    except Exception as e:
        print(f"❌ Error loading {img_path}: {e}")

if not known_face_encodings:
    print("❌ No valid employee profiles loaded. Please add a clear face image to the 'employees' folder.")
    exit()

# 2. Start video capture from camera (0 is usually the built-in webcam)
video_capture = cv2.VideoCapture(0)

print("\nStarting camera stream... Press 'q' on your keyboard to exit.")

while True:
    ret, frame = video_capture.read()
    if not ret:
        print("❌ Failed to grab frame from camera.")
        break
    
    # Resize frame to 1/4 size for faster processing
    small_frame = cv2.resize(frame, (0, 0), fx=0.25, fy=0.25)
    
    # Convert from BGR color (OpenCV) to RGB color (face_recognition)
    rgb_small_frame = cv2.cvtColor(small_frame, cv2.COLOR_BGR2RGB)
    
    # Find all faces and encodings in the current resized frame
    face_locations = face_recognition.face_locations(rgb_small_frame)
    face_encodings = face_recognition.face_encodings(rgb_small_frame, face_locations)
    
    for (top, right, bottom, left), face_encoding in zip(face_locations, face_encodings):
        matches = face_recognition.compare_faces(known_face_encodings, face_encoding)
        name = "Unknown"
        
        # Calculate face distances to find the closest matching face (highest accuracy match)
        face_distances = face_recognition.face_distance(known_face_encodings, face_encoding)
        if len(face_distances) > 0:
            best_match_index = np.argmin(face_distances)
            if matches[best_match_index]:
                name = known_face_names[best_match_index]

        # Scale back up face locations since the frame we detected in was scaled to 1/4 size
        top *= 4
        right *= 4
        bottom *= 4
        left *= 4

        # Draw a bounding box around the detected face
        cv2.rectangle(frame, (left, top), (right, bottom), (0, 255, 0), 2)
        # Draw a text label below the face
        cv2.rectangle(frame, (left, bottom - 35), (right, bottom), (0, 255, 0), cv2.FILLED)
        cv2.putText(frame, name, (left + 6, bottom - 6), cv2.FONT_HERSHEY_DUPLEX, 0.6, (0, 0, 0), 1)

    # Display the live window feed
    cv2.imshow('CCTV Access Verification Feed', frame)

    # Hit 'q' on the keyboard to safely quit the script
    if cv2.waitKey(1) & 0xFF == ord('q'):
        break

# Clean up and close windows
video_capture.release()
cv2.destroyAllWindows()
print("Camera stream closed successfully.")
